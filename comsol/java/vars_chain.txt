kap_t	if(mot==0, (t>=300[s])*(t<600[s])*sin(2*pi*f_walk*(t-300[s]))/R_walk + (t>=600[s])*(t<900[s])*0.5*(1-cos(2*pi*f_flex*(t-600[s])))/R_wrist + (t>=900[s])*(t<1200[s])/R_hold + (t>=1200[s])*(t<1500[s])*0.5*(1-cos(2*pi*f_flex*(t-1200[s])))/R_elbow, (mot==1)*0.5*(1-cos(2*pi*f_flex*t))/R_wrist + (mot==2)*0.5*(1-cos(2*pi*f_flex*t))/R_elbow + (mot==3)*sin(2*pi*f_walk*t)/R_walk + (mot==4)/R_hold)
kap	kap_s + kap_t
eps	epk*kap + eps_s
GF_ch	GF_min + (GF_max-GF_min)*exp(-(V_ov-V0_gf)/V_gf)
GF_c	-dPhiB_ps/kT
GF_tot	(1-r_c)*GF_ch + r_c*GF_c

xtalk	1 + kappa_x*eps
lambda_D	0.304[nm]/sqrt(I_ion/1[M])

FS	dVT_max/V_ov

th_app	max(R_sens*V_ov/dVT_max, 0)
sat	th_app>=1

resid_twin_noX	abs(dI_strain*mism)/FS
xt_floor	kappa_x*eps*R_bio0/FS
tau_on	1/(k_on*Cb_M + k_off)
